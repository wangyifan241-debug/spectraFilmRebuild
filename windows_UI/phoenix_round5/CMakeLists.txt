cmake_minimum_required(VERSION 3.22)
project(PhoenixRound5 LANGUAGES C CXX)
include(CTest)
get_filename_component(SPEKTRALAB_SOURCE_ROOT "../.." ABSOLUTE)
set(SPEKTRALAB_SOURCE_ROOT "${SPEKTRALAB_SOURCE_ROOT}" CACHE PATH "" FORCE)
set(SPEKTRALAB_BUILD_NATIVE_RAW ON CACHE BOOL "" FORCE)
set(SPEKTRALAB_BUILD_DESKTOP ON CACHE BOOL "" FORCE)
add_subdirectory("${SPEKTRALAB_SOURCE_ROOT}/engine" engine)
target_compile_definitions(spektralab_engine PRIVATE SPK_ENABLE_ROUND4_RESEARCH=1)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2)
qt_standard_project_setup(REQUIRES 6.8)
add_subdirectory(.. desktop)
qt_add_executable(spk_phoenix_round5 main.cpp)
target_link_libraries(spk_phoenix_round5 PRIVATE spk_material_library Qt6::Quick)
set_target_properties(spk_phoenix_round5 PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/app")
add_dependencies(spk_phoenix_round5 spektralab_qt_resources)
set_property(TARGET spk_phoenix_round5 APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/scan.comp;${CMAKE_CURRENT_SOURCE_DIR}/Viewer.qml")
add_custom_command(TARGET spk_phoenix_round5 POST_BUILD
 COMMAND "${SPEKTRALAB_GLSLANG}" -V "${CMAKE_CURRENT_SOURCE_DIR}/scan.comp" -o "${CMAKE_BINARY_DIR}/app/resources/vulkan/spk_round4_scan.spv"
 COMMAND "${CMAKE_COMMAND}" -E copy "${CMAKE_CURRENT_SOURCE_DIR}/Viewer.qml" "${CMAKE_BINARY_DIR}/app/Viewer.qml")
if(BUILD_TESTING)
 find_program(ROUND4_NODE node REQUIRED)
 add_test(NAME spk_round4_dataset COMMAND "${ROUND4_NODE}" --test "${SPEKTRALAB_SOURCE_ROOT}/validation/phoenix-round4/tests.cjs")
 add_test(NAME spk_round4_spectral_reference COMMAND "${ROUND4_NODE}" "${SPEKTRALAB_SOURCE_ROOT}/validation/phoenix-round4/scan-reference.cjs" --test)
 add_test(NAME spk_round4_gpu_scan COMMAND "${ROUND4_NODE}" "${SPEKTRALAB_SOURCE_ROOT}/validation/phoenix-round4/run-tests.cjs" "$<TARGET_FILE:spk_phoenix_round5>" "${CMAKE_BINARY_DIR}/app/resources" "${CMAKE_BINARY_DIR}")
 set_tests_properties(spk_round4_gpu_scan PROPERTIES TIMEOUT 240)
endif()
