cmake_minimum_required(VERSION 3.22)
project(SpektraLabIcon LANGUAGES CXX)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Svg)
add_executable(spk_generate_icon main.cpp)
target_compile_features(spk_generate_icon PRIVATE cxx_std_20)
target_link_libraries(spk_generate_icon PRIVATE Qt6::Core Qt6::Gui Qt6::Svg)
add_executable(spk_verify_icon ../../tests/application_icon.cpp)
target_compile_features(spk_verify_icon PRIVATE cxx_std_20)
target_link_libraries(spk_verify_icon PRIVATE Qt6::Core Qt6::Gui)
