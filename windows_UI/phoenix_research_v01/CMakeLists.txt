cmake_minimum_required(VERSION 3.22)
project(PhoenixResearchV01 LANGUAGES C CXX)
include(CTest)
get_filename_component(SPEKTRALAB_SOURCE_ROOT "../.." ABSOLUTE)
set(SPEKTRALAB_SOURCE_ROOT "${SPEKTRALAB_SOURCE_ROOT}" CACHE PATH "" FORCE)
set(SPEKTRALAB_BUILD_NATIVE_RAW ON CACHE BOOL "" FORCE)
set(SPEKTRALAB_BUILD_DESKTOP ON CACHE BOOL "" FORCE)
add_subdirectory("${SPEKTRALAB_SOURCE_ROOT}/engine" engine)
target_compile_definitions(spektralab_engine PRIVATE SPK_ENABLE_ROUND4_RESEARCH=1)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2)
qt_standard_project_setup(REQUIRES 6.8)
add_subdirectory(.. desktop)
qt_add_executable(spk_phoenix_research_v01 main.cpp)
target_link_libraries(spk_phoenix_research_v01 PRIVATE spk_material_library Qt6::Quick)
set_target_properties(spk_phoenix_research_v01 PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/app")
add_dependencies(spk_phoenix_research_v01 spektralab_qt_resources)
set_property(TARGET spk_phoenix_research_v01 APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../phoenix_round5/scan.comp;${CMAKE_CURRENT_SOURCE_DIR}/Viewer.qml")
add_custom_command(TARGET spk_phoenix_research_v01 POST_BUILD
 COMMAND "${SPEKTRALAB_GLSLANG}" -V "${CMAKE_CURRENT_SOURCE_DIR}/../phoenix_round5/scan.comp" -o "${CMAKE_BINARY_DIR}/app/resources/vulkan/spk_round4_scan.spv"
 COMMAND "${CMAKE_COMMAND}" -E copy "${CMAKE_CURRENT_SOURCE_DIR}/Viewer.qml" "${CMAKE_BINARY_DIR}/app/Viewer.qml")
